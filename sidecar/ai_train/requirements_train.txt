# Dependencies for the Unsloth train image (Gemma-4 4B and 12B: train + inference). Installed by the
# provider at image build; shared by the 4B and 12B functions.
#
# Modernised stack: unsloth 2026.10.x supports transformers<=5.17 (incl. gemma4_unified),
# trl<=1.15, peft>=0.18. The old pins (transformers<=5.5, torch<2.9, torchao<0.14) are gone:
# peft now rejects torchao<0.16, and torchao>=0.16 needs a recent torch.
# unsloth/unsloth_zoo are pinned to one release (they must match); everything else is
# bounded by unsloth's own requirements, so pip's resolver picks a compatible set.
# extra-index-url gives the CUDA 12.8 torch wheels; it must match the base image
# (nvidia/cuda:12.8.1-devel-ubuntu22.04, see providers/modal_provider.py).

--extra-index-url https://download.pytorch.org/whl/cu128

modal>=0.63.0                       # provider SDK: volume commit/reload inside the container
torch>=2.10.0,<2.15.0
torchvision

unsloth==2026.10.6
unsloth_zoo==2026.10.6

transformers>=5.5.0,<=5.17.0        # SFTTrainer/model loading; upper bound = unsloth's
trl>=0.24.0,<=1.15.0                # SFTTrainer
datasets>=3.4.1,<5.0.0,!=4.4.*,!=4.5.0
peft>=0.18.0                        # LoRA adapters
accelerate>=1.0.0
bitsandbytes>=0.48.0                # 4-bit quantization
torchao>=0.16.0                     # peft requires >=0.16 whenever it probes for torchao
